@app.route("/auth/roblox/callback")
def roblox_callback():
    code = request.args.get("code")
    if not code:
        return "Virhe: Kirjautumiskoodia ei löytynyt.", 400

    payload = {
        "grant_type": "authorization_code",
        "client_id": CLIENT_ID,
        "client_secret": CLIENT_SECRET,
        "code": code,
        "redirect_uri": REDIRECT_URI
    }
    response = requests.post("https://roblox.com", data=payload, headers={"Content-Type": "application/x-www-form-urlencoded"})
    if response.status_code != 200:
        return f"Virhe tokenin haussa: {response.text}", 400

    access_token = response.json().get("access_token")
    user_response = requests.get("https://roblox.com", headers={"Authorization": f"Bearer {access_token}"})
    if user_response.status_code != 200:
        return "Virhe käyttäjätietojen haussa.", 400

    user_info = user_response.json()
    roblox_id = str(user_info.get("sub"))
    username = user_info.get("preferred_username", "Käyttäjä")

    user = User.query.filter_by(roblox_id=roblox_id).first()
    if not user:
        is_first = User.query.count() == 0
        user = User(
            roblox_id=roblox_id,
            username=username,
            role="Omistaja" if is_first else "Poliisi",
            department="Ylläpito" if is_first else "Poliisi",
            rank="Päällikkö" if is_first else "Vanhempi konstaapeli"
        )
        db.session.add(user)
        db.session.commit()

    session["user_id"] = user.id
    session["username"] = user.username
    log_action(user.username, "Kirjautui sisään Fin CAD -järjestelmään")
    return redirect("/")

@app.route("/api/calls", methods=["GET", "POST"])
def handle_calls():
    if "user_id" not in session:

    if request.method == "POST":
        data = request.json
        call_num = f"112-{random.randint(1000, 9999)}"
        db.session.add(Call(
            call_number=call_num,
            caller=data.get("caller", session["username"]),
            location=data.get("location", "Tuntematon"),
            call_type=data.get("call_type", "Yleinen tehtävä"),
            priority=data.get("priority", "Kiireellinen (A)"),
            description=data.get("description", "")
        ))
        db.session.commit()

        return jsonify({"status": "ok"})

    calls = Call.query.filter(Call.status != "Suljettu").order_by(Call.id.desc()).all()
    return jsonify([{
        "id": c.id, "call_number": c.call_number, "caller": c.caller,
        "location": c.location, "call_type": c.call_type,
        "priority": c.priority, "description": c.description, "status": c.status
